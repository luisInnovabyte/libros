6. Tipos de Datos

              # NUMÉRICOS
              # tinyint, boolean, smallint, mediumint, int, bigint


#tinyint: DE -128 A 127 (Unsigned DE 0 a 255)
#boolean, bool: El 0 es FALSE, cualquier otro número entero es TRUE
#smallint: DE -32768 a 32767 (Unsigned de 0 a 65535)
#Mediumint: de -8388608 a 8388607 (Unsigned de 0 a 16777215)
#int: de -2147483648 a 2147483647 (Unsigned de 0 a 4294967295)
#bigint: de -9223372036854775808 a 9223372036854775807 (Unsigned de 0 a 18446744073709551615)



                #DECIMAL, FLOAT y DOUBLE


#DECIMAL: M,D - (M) es el número total de dígitos, incluidos los decimales, mientras que (D) son la cantidad de dígitos que van detras de la coma decimal
# ZEROFILL , se rellenará con 0 hasta el valor de M
# Si se permiten negativos el simbolo - no cuenta para M
#FLOAT: M,D - (M) es el número total de dígitos, incluidos los decimales, mientras que (D) son la cantidad de dígitos que van detras de la coma decimal
# Utiliza 32 bits para la precisión
#DOUBLE: M,D - (M) es el número total de dígitos, incluidos los decimales, mientras que (D) son la cantidad de dígitos que van detras de la coma decimal
# Utiliza 64 bits para la precisión



                # CADENAS
                # CHAR, VARCHAR, TEXT, ENUM


#CHAR:- Es una cadena de longitud fija de hasta 255 caracteres, al almacenar se rellena con espacios hasta la longitud especificada.
        # Optimo para Si/NO, M/F (Masculino/Femenino).ALTER
        # Cuando se recupera se eliminan los caracteres en blanco.
#VARCHAR- Es una cadena de longitud variable (a la hora de guardar se va a limitar a guardar los datos que realmente lleguen, independientemente de la M que se le haya colocado).
#TEXT - En este caso es opcional y el valor máximo es de 65.535 caracteres. (El M no se suele utilizar).
#MEDIUMTEXT- Es identico a TEXT pero de mayor capacidad.

7. Ejemplos de Consultas

# Leer todos los datos de una tabla con todas sus columna
SELECT * FROM copia_empleados;
# solo una columna
SELECT apellido FROM copia_empleados;
# Todas las columnas trayendo varios campos
SELECT DISTINCT nombre FROM copia_empleados;

7.1 Ejemplos de Consultas - Alias

#Nos permite especificar los nombres de las cabeceras de los campos sin varias la definición de las tablas
SELECT nombre as NOMBRE, apellido as APELLIDO FROM copia_empleados;

7.2 Ejemplos de Consultas - Order by - W3SCHOOLS 

#Nos permite especificar el orden en los que va a mostrar los datos traidos
SELECT * FROM copia_empleados ORDER BY id_empleados ASC;
SELECT * FROM copia_empleados ORDER BY apellido, id_empleados DESC;

7.3 Ejemplos de Consultas - LIMIT - W3SCHOOLS 

#Nos permite especificar la cantidad de registros que nos va a traer
SELECT * FROM copia_empleados ORDER BY nombre LIMIT 3;

7.4 Ejemplos de Consultas - WHERE

# Condicionar los registros que va a traer una tabla
SELECT * FROM copia_empleados WHERE nombre="Luis Carlos";

7.5 Ejemplos de Consultas - DISTINC


# Elimina la vista de los registros duplicados
# Va siempre detrás del SELECT, tiene en cuenta los campos que nos traemos, porEjem. En el siguiente caso está
comparando todos los campos
SELECT DISTINCT * FROM copia_empleados;
# En el caso siguiente nos comparará solo los campos nombre.
SELECT DISTINCT nombre FROM copia_empleados;

8.6 IN, NOT IN

#IN - Nos permite traernos los registros que cumplan la condicion de estar en una lista que le pasamos. En este caso nos
lo traerá si están en la lista.
#NOT IN - Nos traerá los registros que no están en la lista.

# Vamos a traernos los registros que están en la lista. (IN) - Me va a traer los dos registros solcitados
SELECT * FROM ciudades WHERE nombre in ('Armenia', 'Bogota');
                
#Quiero que me traigas todos menos los dos mencionados.
SELECT * FROM ciudades WHERE nombre NOT IN ('Armenia', 'Bogotá');

#GROUP BY
#La declaración GROUP BY agrupa filas que tienen los mismos valores en filas de resumen,
#como "encontrar la cantidad de clientes en cada país".
#La declaración GROUP BY se usa a menudo con funciones agregadas (COUNT(), MAX(), MIN(), SUM(), AVG())
#para agrupar el conjunto de resultados por una o más columnas.

SELECT pais, count(pais) FROM clientes GROUP BY pais ORDER BY pais ;

SELECT pais, sum(importe) as importe FROM clientes GROUP BY pais having importe>90000 order by importe ASC;

total de los importes.
SELECT pais, sum(importe) as TOTAL FROM clientes GROUP BY pais WITH ROLLUP;

CREATE EVENT insertion_event

   ENDS '2018-07-07 19:00:00'

SHOW events\G;

ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 1 MINUTE
ON COMPLETION PRESERVE

    CREATE DATABASE IF NOT EXISTS prueba;

   CREATE TABLE clientes (
    id_cliente INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
    nombre varchar(100),
    fec_nuevo DATETIME
    ) ENGINE = 'INNODB';
   
    CREATE TABLE IF NOT EXISTS articulos (
    id_articulo INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    descr_articulo VARCHAR(100),
    precio_articulo float(8,2),
    fec_nuevo DATETIME
    ) ENGINE='INNODB';
   
    CREATE TABLE IF NOT EXISTS ordenes (
    id_ordenes INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    cliente_id INT NOT NULL,
    articulo_id INT NOT NULL,
    nombre varchar(100) DEFAULT 'Sin nombre',
    cantidad INT,
    kk varchar(100),
    fecha DATE,
    CONSTRAINT FK_CLIENTE FOREIGN KEY (cliente_id) REFERENCES clientes(id_cliente) ON UPDATE NO ACTION ON DELETE NO ACTION,
    CONSTRAINT FK_ARTICULO FOREIGN KEY (articulo_id) REFERENCES articulos(id_articulo) ON UPDATE NO ACTION ON DELETE NO ACTION
    ) ENGINE = 'INNODB';

            #VISTAS PARA COLOCAR LA FECHA DE CLIENTES AL DERECHO
    CREATE OR REPLACE VIEW v_clientes AS (
    SELECT id_cliente, nombre, DATE_FORMAT(fec_nuevo, '%d-%m-%Y') as fecha, DATE_FORMAT(fec_nuevo,
    '%H:%i:%S') as hora
    FROM clientes
    );
   
    
            #VISTAS PARA COLOCAR LA FECHA DE ARTICULOS AL DERECHO
    CREATE OR REPLACE VIEW v_articulos AS (
    SELECT
    id_articulo,
    CONCAT(UPPER(left(ltrim(descr_articulo),1)), SUBSTRING(ltrim(descr_articulo),2,
    CHAR_LENGTH(ltrim(descr_articulo)))) as
    descr_articulo,
    precio_articulo,
    DATE_FORMAT(fec_nuevo, '%d-%m-%Y %h:%i:%s') as fec_nuevo
    FROM articulos
    );
   
    
            #VISTAS DE ORDENES
    CREATE OR REPLACE VIEW v_ordenes AS (
    SELECT
    c.id_cliente, c.nombre as nombre_cliente,
    a.id_articulo, a.descr_articulo, a.precio_articulo,
    o.cantidad , o.articulo_id, o.cliente_id
    FROM
    (ordenes o left join articulos a on o.articulo_id=a.id_articulo)
    left join clientes c on id_cliente=cliente_id
    );
   
    
            ## Vamos a agrupar las cantidades de Euros que han consumido cada cliente.
    select id_cliente, nombre_cliente, sum(precio_articulo*cantidad) as gasto from v_ordenes GROUP
    BY id_cliente,
    nombre_cliente;
   
    
            #Vamos a calcular la media de precio que han consumido por cliente
    select avg(precio_articulo*cantidad) as media from v_ordenes;
   select * from
    (select
    id_cliente,
    nombre_cliente,
    sum(precio_articulo*cantidad) as gasto
    from v_ordenes
    GROUP BY id_cliente, nombre_cliente) as orde
    WHERE orde.gasto >= (select avg(precio_articulo*cantidad) as media from v_ordenes);
