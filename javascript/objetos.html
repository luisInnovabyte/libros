Objetos en JavaScript

Los objetos son una de las estructuras de datos más importantes en JavaScript. Permiten agrupar datos relacionados y funcionalidades en una sola entidad.

💡 Concepto clave: Un objeto es una colección de propiedades, donde cada propiedad es un par clave-valor. Los valores pueden ser primitivos, otros objetos o funciones (métodos).

¿Qué son los objetos?

Los objetos son estructuras de datos que nos permiten:

  • Agrupar datos relacionados: Como propiedades de una persona (nombre, edad, email)
  • Modelar entidades del mundo real: Usuarios, productos, configuraciones
  • Organizar código: Métodos relacionados con los datos
  • Crear estructuras complejas: Objetos dentro de objetos

⚡ Sintaxis Básica

Aprende las diferentes formas de crear y definir objetos en JavaScript.

🔧 Literal de Objeto (Recomendado)

// Sintaxis básica
const objeto = {
    clave: "valor",
    numero: 42,
    activo: true
};

// Ejemplo real
const persona = {
    nombre: "Juan",
    edad: 30,
    email: "juan@example.com",
    activo: true
};

console.log(persona.nombre); // "Juan"

🏗️ Constructor Object()

// Usando constructor
const persona = new Object();
persona.nombre = "Juan";
persona.edad = 30;
persona.email = "juan@example.com";

// O con propiedades iniciales
const config = new Object({
    tema: "claro",
    idioma: "es"
});

console.log(persona.edad); // 30
⚠️ Recomendación: Usa la sintaxis literal { } siempre que sea posible. Es más clara, concisa y eficiente.

🔑 Acceso a Propiedades

Dos métodos principales para acceder y manipular las propiedades de un objeto.

📍 Notación de Punto (.)

const usuario = {
    nombre: "Ana",
    edad: 25,
    email: "ana@example.com"
};

// Leer propiedades
console.log(usuario.nombre);  // "Ana"
console.log(usuario.edad);    // 25

// Modificar propiedades
usuario.edad = 26;
usuario.activo = true;  // Añadir nueva

console.log(usuario.edad);    // 26
console.log(usuario.activo);  // true

✅ Usa cuando conozcas el nombre de la propiedad

🔧 Notación de Corchetes ([])

const usuario = {
    nombre: "Ana",
    "apellido completo": "García López",
    edad: 25
};

// Leer con corchetes
console.log(usuario["nombre"]);           // "Ana"
console.log(usuario["apellido completo"]); // "García López"

// Con variables
const propiedad = "edad";
console.log(usuario[propiedad]);  // 25

// Propiedades dinámicas
const prefijo = "user";
usuario[prefijo + "_id"] = 123;
console.log(usuario.user_id);     // 123

✅ Usa con espacios, variables o propiedades dinámicas

            }
        },
        {
            id: 2,

// Con optional chaining (ES2020)

console.log(empresa.departamento?.rrhh?.jefe);  // undefined (sin error)

// Sin optional chaining (puede dar error)
// console.log(empresa.departamento.rrhh.jefe);  // ❌ Error!

// Verificación manual
if (empresa.departamento && empresa.departamento.rrhh) {
    console.log(empresa.departamento.rrhh.jefe);
}

🔧 Métodos

Los métodos son funciones que pertenecen a un objeto y pueden acceder a sus propiedades usando this.

📝 Sintaxis Tradicional

const calculadora = {
    valor: 0,
    
    sumar: function(numero) {
        this.valor += numero;
        return this; // Para encadenamiento
    },
    
    restar: function(numero) {
        this.valor -= numero;
        return this;
    },
    
    obtenerResultado: function() {
        return this.valor;
    }
};

// Uso
calculadora.sumar(10).restar(3).sumar(5);
console.log(calculadora.obtenerResultado()); // 12

⚡ Sintaxis ES6+ (Recomendada)

const persona = {
    nombre: "Ana",
    edad: 25,
    
    // Método ES6
    saludar() {
        return `Hola, soy ${this.nombre}`;
    },
    
    cumplirAnos() {
        this.edad++;
        return `Ahora tengo ${this.edad} años`;
    },
    
    // Getter
    get info() {
        return `${this.nombre}, ${this.edad} años`;
    },
    
    // Setter
    set nuevaEdad(edad) {
        if (edad > 0) {
            this.edad = edad;
        }
    }
};

console.log(persona.saludar());     // "Hola, soy Ana"
console.log(persona.info);          // "Ana, 25 años"
persona.nuevaEdad = 26;             // Usar setter
⚠️ Cuidado con Arrow Functions: No uses arrow functions para métodos si necesitas acceder a this, ya que no tienen su propio contexto.

✅ Verificación de Propiedades

Métodos para comprobar si una propiedad existe en un objeto.

🔍 Operador 'in'

const usuario = {
    nombre: "Juan",
    edad: 30,
    email: undefined
};

// Verifica si la propiedad existe (incluso si es undefined)
console.log("nombre" in usuario);     // true
console.log("edad" in usuario);       // true  
console.log("email" in usuario);      // true (existe pero es undefined)
console.log("telefono" in usuario);   // false

// También funciona con propiedades heredadas
console.log("toString" in usuario);   // true (heredada de Object)

🔑 hasOwnProperty()

const usuario = {
    nombre: "Juan",
    edad: 30
};

// Solo propiedades propias (no heredadas)
console.log(usuario.hasOwnProperty("nombre"));    // true
console.log(usuario.hasOwnProperty("edad"));      // true
console.log(usuario.hasOwnProperty("toString"));  // false

// Forma más segura
console.log(Object.prototype.hasOwnProperty.call(usuario, "nombre")); // true

📊 Comparación undefined

const config = {
    tema: "claro",
    debug: false,
    version: undefined
};

// Verificar si tiene valor (no undefined)
console.log(config.tema !== undefined);     // true
console.log(config.debug !== undefined);    // true  
console.log(config.version !== undefined);  // false
console.log(config.inexistente !== undefined); // false

// Cuidado: false, 0, "", null son valores válidos
console.log(config.debug !== undefined);    // true (false es válido)

🎯 Método personalizado

function tienePropiedad(objeto, propiedad) {
    return propiedad in objeto && objeto[propiedad] !== undefined;
}

function tienePropiedadConValor(objeto, propiedad) {
    return propiedad in objeto && 
           objeto[propiedad] !== undefined && 
           objeto[propiedad] !== null;
}

const datos = {
    nombre: "Ana",
    edad: null,
    activo: false
};

console.log(tienePropiedad(datos, "nombre"));  // true
console.log(tienePropiedad(datos, "edad"));    // false (es null)
console.log(tienePropiedadConValor(datos, "activo")); // true

🔄 Modificación de Objetos

Diferentes formas de modificar, actualizar y extender objetos.

➕ Añadir/Modificar Propiedades

const usuario = {
    nombre: "Ana",
    edad: 25
};

// Añadir nuevas propiedades
usuario.email = "ana@example.com";
usuario.activo = true;

// Modificar existentes
usuario.edad = 26;

// Propiedades dinámicas
const propiedad = "ultimoAcceso";
usuario[propiedad] = new Date();

// Con template literals
const prefijo = "user";
usuario[`${prefijo}_id`] = 12345;

console.log(usuario);
// {
//   nombre: "Ana",
//   edad: 26,
//   email: "ana@example.com",
//   activo: true,
//   ultimoAcceso: 2023-12-07T10:30:00.000Z,
//   user_id: 12345
// }

🗑️ Eliminar Propiedades

const usuario = {
    nombre: "Ana",
    edad: 25,
    email: "ana@example.com",
    temporal: "borrar",
    debug: true
};

// Eliminar con delete
delete usuario.temporal;
delete usuario.debug;

console.log(usuario);
// {
//   nombre: "Ana", 
//   edad: 25,
//   email: "ana@example.com"
// }

// Verificar que se eliminó
console.log("temporal" in usuario); // false

// Eliminar múltiples (con destructuring)
const { debug, temporal, ...usuarioLimpio } = usuario;
console.log(usuarioLimpio); // Sin debug ni temporal
⚠️ Performance: El operador delete puede ser lento. Para mejor rendimiento, considera crear un nuevo objeto sin las propiedades no deseadas.

📋 Copia vs Referencia

Comprende la diferencia entre asignación por referencia y copia de objetos.

⚠️ Importante: Los objetos se asignan por referencia, no por valor. Esto significa que múltiples variables pueden apuntar al mismo objeto en memoria.

🔗 Asignación por Referencia

const original = {
    nombre: "Juan",
    edad: 30
};

// Asignación por referencia
const referencia = original;
referencia.edad = 31;

console.log(original.edad);    // 31 (¡se modificó!)
console.log(referencia.edad);  // 31

// Ambas variables apuntan al mismo objeto
console.log(original === referencia); // true

❌ Modifica el objeto original

📄 Copia Superficial

const original = {
    nombre: "Juan",
    edad: 30,
    activo: true
};

// Copia superficial con spread
const copia1 = { ...original };

// Copia superficial con Object.assign
const copia2 = Object.assign({}, original);

copia1.edad = 32;
console.log(original.edad);  // 30 (no se modificó)
console.log(copia1.edad);    // 32

// Son objetos diferentes
console.log(original === copia1); // false

✅ Crea una copia independiente (primer nivel)

⚠️ Cuidado con objetos anidados: La copia superficial solo copia el primer nivel. Los objetos anidados siguen siendo referencias. 
const complejo = {
    usuario: {
        nombre: "Ana",
        configuracion: {
            tema: "oscuro"
        }
    },
    activo: true
};

// Copia superficial
const copiaSuperficial = { ...complejo };
copiaSuperficial.usuario.nombre = "María"; // ¡Modifica el original!

console.log(complejo.usuario.nombre); // "María" (se modificó)

// Para copia profunda (objetos simples sin funciones/fechas)
const copiaProfunda = JSON.parse(JSON.stringify(complejo));
copiaProfunda.usuario.nombre = "Laura"; // No modifica el original

console.log(complejo.usuario.nombre);     // "María" (no se modificó)
console.log(copiaProfunda.usuario.nombre); // "Laura"

    return {
        ...estadoActual,

    };

function actualizarConfiguracion(estadoActual, nuevaConfig) {
    return {
        ...estadoActual,

            ...estadoActual.configuracion,
            ...nuevaConfig
        }
    };
}

    cargando: false
});

    tema: "oscuro"

📝 Manejo de Formularios

const formulario = {
    campos: {
        nombre: "",
        email: "",
        edad: ""
    },
    errores: {},
    valido: false
};

// Función para actualizar campo
function actualizarCampo(form, campo, valor) {
    return {
        ...form,
        campos: {
            ...form.campos,
            [campo]: valor
        }
    };
}

// Función para validar
function validarFormulario(form) {
    const errores = {};
    
    if (!form.campos.nombre) {
        errores.nombre = "Nombre es requerido";
    }
    
    if (!form.campos.email.includes("@")) {
        errores.email = "Email inválido";
    }
    
    return {
        ...form,
        errores,
        valido: Object.keys(errores).length === 0
    };
}

// Uso
let formActual = formulario;
formActual = actualizarCampo(formActual, "nombre", "Juan");
formActual = actualizarCampo(formActual, "email", "juan@example.com");
formActual = validarFormulario(formActual);

💾 Sistema de Cache

const cache = {};

// Añadir al cache con expiración
function agregarAlCache(clave, valor, expiracion = null) {
    cache[clave] = {
        valor,
        timestamp: Date.now(),
        expiracion: expiracion ? Date.now() + expiracion : null
    };
}

// Obtener del cache
function obtenerDelCache(clave) {
    const entrada = cache[clave];
    if (!entrada) return null;
    
    // Verificar expiración
    if (entrada.expiracion && Date.now() > entrada.expiracion) {
        delete cache[clave];
        return null;
    }
    
    return entrada.valor;
}

// Limpiar cache expirado
function limpiarCacheExpirado() {
    const ahora = Date.now();
    Object.keys(cache).forEach(clave => {
        const entrada = cache[clave];
        if (entrada.expiracion && ahora > entrada.expiracion) {
            delete cache[clave];
        }
    });
}

// Uso
agregarAlCache("usuario_123", { nombre: "Ana" }, 300000); // 5 min
agregarAlCache("configuracion", { tema: "oscuro" });

const usuario = obtenerDelCache("usuario_123");
console.log(usuario); // { nombre: "Ana" } o null si expiró
