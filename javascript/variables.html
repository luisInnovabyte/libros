Variables en JavaScript

Las variables en JavaScript se utilizan para almacenar datos que pueden ser usados y modificados a lo largo del código.

⚠️ Importante: Las variables en JavaScript distinguen entre MAYÚSCULAS y minúsculas. nombre y Nombre son variables diferentes.

¿Qué son las variables?

Las variables son contenedores que almacenan valores de datos. Piensa en ellas como cajas etiquetadas donde puedes guardar información para usarla más tarde.

  • Almacenan datos: Números, texto, objetos, etc.
  • Se pueden modificar: Sus valores pueden cambiar durante la ejecución
  • Tienen nombres: Identificadores únicos para acceder a los datos
  • Tienen ámbito: Reglas sobre dónde pueden ser accedidas

📝 Declaración de Variables

JavaScript ofrece tres formas principales de declarar variables, cada una con sus propias características.

// Ejemplos básicos de declaración
var nombre = 'Juan';
let edad = 30;
const PI = 3.1416;

// Declaración sin valor inicial
let mensaje;
console.log(mensaje); // undefined

// Asignación posterior
mensaje = "Hola mundo";
console.log(mensaje); // "Hola mundo"

🔤 var, let y const

Cada palabra clave tiene características específicas que debes conocer para elegir la correcta.

🚫 var (Evitar)

// var - Declaración clásica
var nombre = 'Ana';
var nombre = 'Luis'; // Redeclaración permitida

function ejemplo() {
    var mensaje = 'Hola';
    if (true) {
        var mensaje = 'Adiós'; // Misma variable
    }
    console.log(mensaje); // "Adiós"
}

// Hoisting con var
console.log(variable); // undefined (no error)
var variable = 'valor';
  • ❌ Ámbito de función
  • ❌ Permite redeclaración
  • ❌ Hoisting confuso
  • ❌ No usar salvo compatibilidad

✅ let (Variables mutables)

// let - Para valores que cambian
let contador = 0;
contador = 1; // Reasignación permitida

function ejemplo() {
    let mensaje = 'Hola';
    if (true) {
        let mensaje = 'Adiós'; // Variable diferente
        console.log(mensaje); // "Adiós"
    }
    console.log(mensaje); // "Hola"
}

// Ámbito de bloque
if (true) {
    let temporal = 'existe aquí';
}
// console.log(temporal); // ReferenceError
  • ✅ Ámbito de bloque
  • ✅ No permite redeclaración
  • ✅ Permite reasignación
  • ✅ Usar cuando el valor cambie

🔒 const (Constantes)

// const - Para valores constantes
const PI = 3.1416;
const NOMBRE_APP = 'MiApp';

// Debe asignarse al declarar
const usuario = 'Juan';

// No se puede reasignar
// PI = 3.14; // TypeError

// Ámbito de bloque
if (true) {
    const LOCAL = 'solo aquí';
}
// console.log(LOCAL); // ReferenceError

// Con objetos (contenido mutable)
const config = { tema: 'claro' };
config.tema = 'oscuro'; // ✅ Permitido
  • ✅ Ámbito de bloque
  • ✅ No permite reasignación
  • ✅ Debe inicializarse
  • ✅ Usar por defecto
📋 Regla práctica:
1. Usa const por defecto
2. Cambia a let solo si necesitas reasignar
3. Evita var completamente

🏗️ Objetos con const

Aunque const no permite reasignación, sí permite modificar el contenido de objetos y arrays.

📋 ¿Qué es un objeto?

// Objeto básico
const persona = { 
    nombre: "Luis",
    edad: 30
};

// Objeto más complejo
const estudiante = {
    nombre: "Ana",
    edad: 22,
    activo: true,
    materias: ["JavaScript", "CSS", "HTML"],
    direccion: {
        calle: "Principal 123",
        ciudad: "Madrid"
    },
    saludar: function() {
        return `Hola, soy ${this.nombre}`;
    }
};

console.log(estudiante.nombre); // "Ana"
console.log(estudiante.saludar()); // "Hola, soy Ana"

🔧 Acceso a propiedades

const persona = { 
    nombre: "Luis", 
    edad: 30,
    "nombre completo": "Luis García" 
};

// Notación de punto
console.log(persona.nombre);        // "Luis"
console.log(persona.edad);          // 30

// Notación de corchetes
console.log(persona["nombre"]);     // "Luis"
console.log(persona["nombre completo"]); // "Luis García"

// Acceso dinámico
const propiedad = "edad";
console.log(persona[propiedad]);    // 30
💡 Concepto clave: const protege la referencia al objeto, no su contenido. Es como una caja etiquetada: no puedes cambiar la etiqueta, pero sí el contenido. 
const persona = { nombre: "Luis" };

// ✅ PERMITIDO: Modificar propiedades existentes
persona.nombre = "Carlos";

// ✅ PERMITIDO: Añadir nuevas propiedades
persona.edad = 25;
persona.activo = true;

// ✅ PERMITIDO: Eliminar propiedades
delete persona.activo;

console.log(persona); // { nombre: "Carlos", edad: 25 }

// ❌ ERROR: No se puede reasignar el objeto completo
// persona = {}; // TypeError: Assignment to constant variable

🎯 Tipos de Datos

JavaScript es dinámicamente tipado, las variables pueden contener diferentes tipos de datos.

🔤 Tipos Primitivos

// String (texto)
let texto = "Hola mundo";
let texto2 = 'Con comillas simples';
let texto3 = `Template literal: ${texto}`;

// Number (números)
let entero = 42;
let decimal = 3.14;
let negativo = -15;
let infinito = Infinity;

// Boolean (verdadero/falso)
let esVerdadero = true;
let esFalso = false;

// Undefined (sin valor)
let indefinido;
let tambienIndefinido = undefined;

// Null (valor nulo intencionado)
let nulo = null;

// Symbol (ES6+)
let simbolo = Symbol('id');

// BigInt (ES2020+)
let enteroGrande = 123n;

📦 Tipos de Referencia

// Object (objetos)
let persona = { 
    nombre: "Ana",
    edad: 25
};

// Array (arreglos)
let numeros = [1, 2, 3, 4, 5];
let mixto = ["texto", 42, true, null];

// Function (funciones)
let saludo = function() { 
    return "Hola"; 
};

// Date (fechas)
let ahora = new Date();
let cumpleanos = new Date('1990-05-15');

// RegExp (expresiones regulares)
let patron = /[a-z]+/g;

// Verificar tipos
console.log(typeof "texto");    // "string"
console.log(typeof 42);         // "number"
console.log(typeof true);       // "boolean"
console.log(typeof undefined);  // "undefined"
console.log(typeof null);       // "object" (peculiaridad JS)
console.log(typeof []);         // "object"
console.log(typeof {});         // "object"

🔄 Coerción de Tipos

JavaScript convierte automáticamente tipos cuando es necesario. Esto puede ser útil pero también confuso.

⚡ Coerción Automática

// Concatenación vs suma
let resultado1 = "5" + 3;      // "53" (string)
let resultado2 = 5 + 3;        // 8 (number)
let resultado3 = "5" + 3 + 2;  // "532" (string)
let resultado4 = 5 + 3 + "2";  // "82" (number + string)

// Conversión a número
let suma = +"5" + 3;           // 8 (number)
let resta = "10" - 3;          // 7 (number)
let multiplicacion = "4" * 2;  // 8 (number)

// Conversión a boolean
let bool1 = !!"texto";         // true
let bool2 = !!0;               // false
let bool3 = !!"";              // false
let bool4 = !!null;            // false

console.log("5" == 5);         // true (con coerción)
console.log("5" === 5);        // false (sin coerción)

🎯 Conversión Explícita

// String → Number
let num1 = Number("42");       // 42
let num2 = parseInt("42px");   // 42
let num3 = parseFloat("3.14"); // 3.14
let num4 = +"123";             // 123

// Number → String
let str1 = String(123);        // "123"
let str2 = (123).toString();   // "123"
let str3 = 123 + "";           // "123"

// Boolean conversions
let bool1 = Boolean("texto");  // true
let bool2 = Boolean("");       // false
let bool3 = Boolean(0);        // false
let bool4 = Boolean(1);        // true

// Valores falsy
console.log(Boolean(false));   // false
console.log(Boolean(0));       // false
console.log(Boolean(""));      // false
console.log(Boolean(null));    // false
console.log(Boolean(undefined)); // false
console.log(Boolean(NaN));     // false
⚠️ Cuidado: La coerción automática puede causar errores inesperados. Usa siempre === en lugar de == y convierte tipos explícitamente cuando sea necesario.

🌐 Ámbito (Scope)

El ámbito determina dónde y cómo puedes acceder a una variable. Son las "reglas de visibilidad" de las variables.

🔧 Ámbito de Función

function ejemploFuncion() {
    var funcionVar = "Visible en toda la función";
    let funcionLet = "También en toda la función";
    
    if (true) {
        var dentroIf = "var: visible fuera del if";
        let bloqueVar = "let: solo en este bloque";
        
        console.log(funcionVar);  // ✅ Accesible
        console.log(funcionLet);  // ✅ Accesible
        console.log(bloqueVar);   // ✅ Accesible
    }
    
    console.log(dentroIf);     // ✅ var es accesible
    // console.log(bloqueVar); // ❌ ReferenceError
}

// console.log(funcionVar);   // ❌ ReferenceError
ejemploFuncion();

🧱 Ámbito de Bloque

// Ámbito de bloque con let y const
if (true) {
    let mensaje = 'Hola!';
    const CONSTANTE = 'No cambio';
    
    console.log(mensaje);    // ✅ "Hola!"
    console.log(CONSTANTE);  // ✅ "No cambio"
}

// console.log(mensaje);    // ❌ ReferenceError
// console.log(CONSTANTE);  // ❌ ReferenceError

// Bucles y ámbito
for (let i = 0; i < 3; i++) {
    let mensaje = `Iteración ${i}`;
    console.log(mensaje);
}
// console.log(i);          // ❌ ReferenceError
// console.log(mensaje);    // ❌ ReferenceError

// Cada iteración tiene su propio ámbito
for (let i = 0; i < 3; i++) {
    setTimeout(() => {
        console.log(i);      // 0, 1, 2 (correcto)
    }, 100);
}

🌍 Ámbito Global

El ámbito global es el contexto principal del programa. Las variables aquí son visibles desde cualquier lugar.

⚠️ Problemas del ámbito global:
  • Contaminación: Demasiadas variables globales
  • Conflictos: Variables con el mismo nombre
  • Variables accidentales: Olvido de declaración

❌ Problemas Comunes

// Contaminación del espacio global
var nombre = "Juan";
var edad = 30;
var activo = true;
var contador = 0;

// Conflictos de nombres
var datos = "Información importante";
// En otro archivo...
var datos = "Otros datos"; // ¡Sobrescribe!

// Variables accidentalmente globales
function procesarDatos() {
    // ¡Olvido declarar la variable!
    resultado = "Procesado"; // Se vuelve global
}

procesarDatos();
console.log(resultado); // "Procesado" - ¡Global!

✨ Buenas Prácticas

Consejos para escribir código más limpio, legible y mantenible.

📝 Nombres Descriptivos

// ❌ Malo: nombres confusos
const d = new Date();
const u = users.filter(u => u.a);
const calc = (x, y) => x * y * 0.1;

// ✅ Bueno: nombres descriptivos
const currentDate = new Date();

const calculateDiscount = (price, quantity) => 
    price * quantity * 0.1;

// Constantes en MAYÚSCULAS
const MAX_RETRY_ATTEMPTS = 3;
const API_BASE_URL = 'https://api.ejemplo.com';
const DEFAULT_THEME = 'light';

// Booleans con prefijos
const isVisible = true;
const hasPermission = false;
const canEdit = user.role === 'admin';

🏗️ Estructura y Organización

// ✅ Prefiere const por defecto
const configuration = { theme: "dark" };
const userList = ["Ana", "Luis"];

// ✅ Solo usa let si necesitas reasignar
let counter = 0;
for (let i = 0; i < 10; i++) {
    counter += i;
}

// ✅ Evita variables globales
(function() {
    const moduleData = "privado";
    
    // Exponer solo lo necesario
    window.MyModule = {
        publicMethod: function() {
            return "público";
        }
    };
})();

// ✅ Declara cerca del uso
function processUser(userId) {
    const user = findUser(userId);    // Cerca de su uso
    
    if (!user) return null;
    
    const permissions = getPermissions(user.role);
    return { user, permissions };
}

⚠️ Errores Comunes

Los errores más frecuentes al trabajar con variables y cómo evitarlos.

🚫 Error 1: Redeclaración

// ❌ Error: Redeclarar con let/const
let mensaje = "Hola";
let mensaje = "Adiós"; // SyntaxError

const usuario = "Ana";
const usuario = "Luis"; // SyntaxError

// ✅ Correcto: Usar nombres diferentes
let mensajeInicial = "Hola";
let mensajeFinal = "Adiós";

const usuarioActual = "Ana";
const usuarioAnterior = "Luis";

🔒 Error 2: Modificar const

// ❌ Error: Reasignar const
const numero = 5;
numero = 10; // TypeError

const lista = [1, 2, 3];
lista = [4, 5, 6]; // TypeError

// ✅ Correcto: Modificar contenido
const configuracion = { tema: "claro" };
configuracion.tema = "oscuro"; // ✅ OK

const numeros = [1, 2, 3];
numeros.push(4); // ✅ OK
numeros[0] = 0;  // ✅ OK

📍 Error 3: Temporal Dead Zone

// ❌ Error: Usar antes de declarar
console.log(variable); // ReferenceError
let variable = "valor";

console.log(constante); // ReferenceError
const constante = "valor";

// ✅ Correcto: Declarar primero
let mensaje = "Hola";
console.log(mensaje); // "Hola"

const nombre = "Ana";
console.log(nombre); // "Ana"

🌐 Error 4: Variables Globales Accidentales

// ❌ Error: Variable global accidental
function calcular() {
    resultado = 10 * 5; // ¡Sin declarar!
}
calcular();
console.log(resultado); // 50 (global)

// ✅ Correcto: Declarar apropiadamente
function calcular() {
    const resultado = 10 * 5;
    return resultado;
}
const total = calcular();
console.log(total); // 50

// ✅ Usar 'use strict' para prevenir
function calcularSeguro() {
    'use strict';
    // resultado = 10 * 5; // ReferenceError
    const resultado = 10 * 5;
    return resultado;
}

📚 Resumen Final

🔑 Conceptos Clave:

  • const por defecto: Usa para valores que no cambiarán
  • let para mutables: Solo cuando necesites reasignar
  • Evita var: Problemas de ámbito y hoisting
  • Ámbito de bloque: let y const respetan bloques { }
  • Objetos mutables: const protege la referencia, no el contenido

✨ Mejores Prácticas:

  • Nombres descriptivos: userAge mejor que a
  • Minimiza globales: Usa módulos y namespaces
  • Declara cerca del uso: No al inicio de la función
  • Usa 'use strict': Previene errores comunes
  • Conversión explícita: Number() mejor que +
