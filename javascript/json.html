📄 JSON en JavaScript

Guía completa sobre JSON - El formato universal de intercambio de datos

🎯 Introducción

JSON (JavaScript Object Notation) es un formato de intercambio de datos ligero y fácil de leer que se ha convertido en el estándar para la comunicación entre aplicaciones web.

📄 Características principales de JSON:

  • 📝 Legible por humanos: Fácil de leer y escribir
  • 🚀 Ligero: Menos verboso que XML
  • 🌐 Universal: Soportado por todos los lenguajes modernos
  • 📊 Estructurado: Permite datos jerárquicos complejos
  • 🔒 Seguro: No ejecuta código, solo datos

📄 ¿Qué es JSON?

JSON es un formato de texto que representa datos estructurados usando una sintaxis derivada de JavaScript. Es completamente independiente del lenguaje pero utiliza convenciones familiares.

// Ejemplo básico de JSON
const datosJSON = `{
    "nombre": "María García",
    "edad": 28,
    "activo": true,
    "direccion": {
        "calle": "Calle Principal 123",
        "ciudad": "Madrid",
        "codigoPostal": "28001"
    },
    "hobbies": ["lectura", "cine", "cocina"],
    "telefono": null
}`;

console.log("📄 JSON como string:", datosJSON);

📝 Sintaxis de JSON

Reglas importantes de sintaxis

✅ JSON válido

{
    "clave": "valor",
    "numero": 123,
    "booleano": true,
    "array": [1, 2, 3],
    "objeto": {
        "anidado": "valor"
    },
    "nulo": null
}

❌ JSON inválido

// Errores comunes:
{
    clave: "valor",          // ❌ Claves sin comillas
    "otro": 'valor',         // ❌ Comillas simples
    "numero": 123,           // ❌ Coma final
    // "comentario": "no"    // ❌ Comentarios
    "funcion": function(){}, // ❌ Funciones
    "undefined": undefined   // ❌ undefined
}
⚠️ Importante: JSON es más estricto que JavaScript. Las claves DEBEN ir entre comillas dobles, no se permiten comentarios, funciones, o undefined.

Parámetro replacer - Filtrar propiedades

const producto = {
    id: 1001,
    nombre: "Laptop Gaming",
    precio: 1299.99,
    precioInterno: 800,      // Campo interno
    stock: 15,
    descripcion: "Laptop de alta gama para gaming",
    fechaCreacion: new Date(),
    activo: true
};

// Filtrar con array de propiedades
const camposPublicos = ["id", "nombre", "precio", "stock", "descripcion", "activo"];
const productoPublico = JSON.stringify(producto, camposPublicos, 2);

console.log("🔒 Solo campos públicos:", productoPublico);

// Filtrar con función replacer
const productoFiltrado = JSON.stringify(producto, (clave, valor) => {
    // Filtrar campos internos
    if (clave.includes("Interno")) {
        return undefined;
    }
    
    // Formatear precio
    if (clave === "precio") {
        return `${valor} EUR`;
    }
    
    // Formatear fechas
    if (valor instanceof Date) {
        return valor.toLocaleDateString("es-ES");
    }
    
    return valor;
}, 2);

console.log("⚙️ Producto filtrado:", productoFiltrado);

Método toJSON() personalizado

// Objetos pueden definir su propia serialización JSON
class Usuario {
    constructor(nombre, email, password) {
        this.id = Math.random().toString(36).substr(2, 9);
        this.nombre = nombre;
        this.email = email;
        this.password = password; // Campo sensible
        this.fechaCreacion = new Date();
        this.activo = true;
    }
    
    // Método personalizado para JSON.stringify
    toJSON() {
        return {
            id: this.id,
            nombre: this.nombre,
            email: this.email,
            // Omitir password por seguridad
            fechaCreacion: this.fechaCreacion.toISOString(),
            activo: this.activo,
            tipo: "Usuario" // Campo adicional
        };
    }
}

const usuario = new Usuario("Ana Martínez", "ana@email.com", "password123");

console.log(JSON.stringify(usuario, null, 2));

Parámetro reviver - Transformar valores

const datosJSON = `{
    "id": 789,
    "nombre": "Pedro González",
    "fechaNacimiento": "1990-03-15T00:00:00.000Z",
    "fechaRegistro": "2023-01-20T14:30:00.000Z",
    "configuracion": {
        "precio_producto": "29.99",
        "descuento_aplicado": "15.5"
    },
    "coordenadas": {
        "latitud": "40.4168",
        "longitud": "-3.7038"
    },
    "activo": "true"
}`;

// Usar reviver para transformar datos durante el parseo
const objetoTransformado = JSON.parse(datosJSON, (clave, valor) => {
    // Convertir strings de fecha a objetos Date
    if (clave.includes("fecha") || clave.includes("Fecha")) {
        return new Date(valor);
    }
    
    // Convertir strings numéricos a números
    if (clave.includes("precio") || clave.includes("descuento")) {
        return parseFloat(valor);
    }
    
    // Convertir coordenadas a números
    if (clave === "latitud" || clave === "longitud") {
        return parseFloat(valor);
    }
    
    // Convertir string "true"/"false" a boolean
    if (clave === "activo") {
        return valor === "true";
    }
    
    return valor;
});

console.log("🔄 Objeto transformado:", objetoTransformado);

Manejo de errores

// Función robusta para parsear JSON
function parsearJSONSeguro(jsonString, valorPorDefecto = null) {
    try {
        // Verificar que no esté vacío
        if (!jsonString || jsonString.trim() === "") {
            console.warn("⚠️ JSON string vacío");
            return valorPorDefecto;
        }
        
        const objeto = JSON.parse(jsonString);
        
        // Verificar que no sea null
        if (objeto === null) {
            console.warn("⚠️ JSON es null");
            return valorPorDefecto;
        }
        
        return objeto;
        
    } catch (error) {
        console.error("❌ Error parseando JSON:", error.message);
        return valorPorDefecto;
    }
}

1. Sistema de configuración

class GestorConfiguracion {
    constructor() {
        this.configuracionPorDefecto = {
            aplicacion: {
                nombre: "Mi App",
                version: "1.0.0",
                modo: "desarrollo"
            },
            interfaz: {
                tema: "claro",
                idioma: "es",
                animaciones: true
            }
        };
        
        this.configuracionActual = null;
    }
    
    // Cargar configuración desde JSON
    cargarConfiguracion(jsonString) {
        try {
            const configuracion = JSON.parse(jsonString, (clave, valor) => {
                // Validar idioma
                if (clave === "idioma" && !["es", "en", "fr"].includes(valor)) {
                    console.warn(`⚠️ Idioma no soportado (${valor}), usando español`);
                    return "es";
                }
                
                // Validar tema
                if (clave === "tema" && !["claro", "oscuro"].includes(valor)) {
                    console.warn(`⚠️ Tema no soportado (${valor}), usando claro`);
                    return "claro";
                }
                
                return valor;
            });
            
            // Fusionar con configuración por defecto
            this.configuracionActual = {
                ...this.configuracionPorDefecto,
                ...configuracion
            };
            
            console.log("✅ Configuración cargada exitosamente");
            return true;
            
        } catch (error) {
            console.error("❌ Error cargando configuración:", error.message);
            this.configuracionActual = { ...this.configuracionPorDefecto };
            return false;
        }
    }
    
    // Guardar configuración como JSON
    guardarConfiguracion() {
        try {
            return JSON.stringify(this.configuracionActual, null, 2);
        } catch (error) {
            console.error("❌ Error guardando configuración:", error.message);
            return null;
        }
    }
}

2. API REST con JSON

class ClienteAPI {
    constructor(baseURL) {
        this.baseURL = baseURL;
        this.headers = {
            "Content-Type": "application/json",
            "Accept": "application/json"
        };
    }
    
    // GET - Obtener datos
    async get(endpoint, parametros = {}) {
        try {
            const url = new URL(endpoint, this.baseURL);
            
            // Agregar parámetros de consulta
            Object.entries(parametros).forEach(([clave, valor]) => {
                if (valor !== null && valor !== undefined) {
                    url.searchParams.append(clave, valor);
                }
            });
            
            const response = await fetch(url.toString(), {
                method: "GET",
                headers: this.headers
            });
            
            return await this.procesarRespuesta(response);
            
        } catch (error) {
            console.error("❌ Error en GET:", error.message);
            throw error;
        }
    }
    
    // POST - Crear datos
    async post(endpoint, datos) {
        try {
            const body = JSON.stringify(datos, (clave, valor) => {
                // Formatear fechas
                if (valor instanceof Date) {
                    return valor.toISOString();
                }
                return valor;
            });
            
            const response = await fetch(new URL(endpoint, this.baseURL), {
                method: "POST",
                headers: this.headers,
                body: body
            });
            
            return await this.procesarRespuesta(response);
            
        } catch (error) {
            console.error("❌ Error en POST:", error.message);
            throw error;
        }
    }
    
    // Procesar respuesta de la API
    async procesarRespuesta(response) {
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        
        const contentType = response.headers.get("content-type");
        if (contentType && contentType.includes("application/json")) {
            return await response.json();
        }
        
        return await response.text();
    }
}

✅ Validación de JSON

class ValidadorJSON {
    static validarEstructura(objeto, esquema) {
        const errores = [];
        
        // Verificar campos requeridos
        if (esquema.requeridos) {
            esquema.requeridos.forEach(campo => {
                if (!(campo in objeto)) {
                    errores.push(`Campo requerido faltante: ${campo}`);
                }
            });
        }
        
        // Verificar tipos de datos
        if (esquema.tipos) {
            Object.entries(esquema.tipos).forEach(([campo, tipoEsperado]) => {
                if (campo in objeto) {
                    const tipoActual = typeof objeto[campo];
                    if (tipoActual !== tipoEsperado) {
                        errores.push(`${campo}: se esperaba ${tipoEsperado}, recibido ${tipoActual}`);
                    }
                }
            });
        }
        
        return {
            valido: errores.length === 0,
            errores: errores
        };
    }
    
    static validarEmail(email) {
        const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        return regex.test(email);
    }
    
    static validarURL(url) {
        try {
            new URL(url);
            return true;
        } catch {
            return false;
        }
    }
}

✅ Buenas prácticas

// 1. Siempre usar try/catch para JSON.parse()
function parsearSeguro(jsonString, valorPorDefecto = null) {
    try {
        return JSON.parse(jsonString);
    } catch (error) {
        console.error("Error parseando JSON:", error.message);
        return valorPorDefecto;
    }
}

// 2. Filtrar datos sensibles en stringify
function serializar(objeto) {
    return JSON.stringify(objeto, (clave, valor) => {
        // Filtrar campos sensibles
        if (clave.includes("password") || clave.includes("secret")) {
            return undefined;
        }
        
        // Formatear fechas
        if (valor instanceof Date) {
            return valor.toISOString();
        }
        
        return valor;
    });
}

// 3. Validar estructura antes de usar
function procesarUsuario(jsonString) {
    const usuario = parsearSeguro(jsonString);
    
    if (!usuario || typeof usuario !== "object") {
        throw new Error("JSON inválido");
    }
    
    if (!usuario.nombre || !usuario.email) {
        throw new Error("Campos requeridos faltantes");
    }
    
    return usuario;
}

🔍 Herramientas de Debugging

class JSONDebugger {
    static validarYMostrarErrores(jsonString) {
        try {
            const objeto = JSON.parse(jsonString);
            console.log("✅ JSON válido:", objeto);
            return objeto;
        } catch (error) {
            console.error("❌ JSON inválido:");
            console.error("Error:", error.message);
            
            // Intentar mostrar dónde está el error
            const match = error.message.match(/position (\d+)/);
            if (match) {
                const posicion = parseInt(match[1]);
                const contexto = jsonString.substring(Math.max(0, posicion - 20), posicion + 20);
                console.error("Contexto del error:", contexto);
                console.error(" ".repeat(20) + "^");
            }
            
            return null;
        }
    }
    
    static formatear(objeto, indentacion = 2) {
        return JSON.stringify(objeto, null, indentacion);
    }
    
    static compararJSONs(json1, json2) {
        const obj1 = typeof json1 === "string" ? JSON.parse(json1) : json1;
        const obj2 = typeof json2 === "string" ? JSON.parse(json2) : json2;
        
        const diferencias = [];
        this.compararRecursivo(obj1, obj2, "", diferencias);
        
        if (diferencias.length === 0) {
            console.log("✅ Los JSONs son idénticos");
        } else {
            console.log("📊 Diferencias encontradas:");
            diferencias.forEach(diff => {
                console.log(`  ${diff.ruta}: ${diff.descripcion}`);
            });
        }
        
        return diferencias.length === 0;
    }
    
    static compararRecursivo(obj1, obj2, ruta, diferencias) {
        if (typeof obj1 !== typeof obj2) {
            diferencias.push({
                ruta: ruta || "raíz",
                descripcion: `Tipo diferente: ${typeof obj1} vs ${typeof obj2}`
            });
            return;
        }
        
        if (obj1 === null || obj2 === null) {
            if (obj1 !== obj2) {
                diferencias.push({
                    ruta: ruta || "raíz",
                    descripcion: `${obj1} vs ${obj2}`
                });
            }
            return;
        }
        
        if (typeof obj1 === "object") {
            const keys1 = Object.keys(obj1);
            const keys2 = Object.keys(obj2);
            
            // Verificar claves diferentes
            const todasLasClaves = new Set([...keys1, ...keys2]);
            
            todasLasClaves.forEach(key => {
                const nuevaRuta = ruta ? `${ruta}.${key}` : key;
                
                if (!(key in obj1)) {
                    diferencias.push({
                        ruta: nuevaRuta,
                        descripcion: "Clave solo en segundo objeto"
                    });
                } else if (!(key in obj2)) {
                    diferencias.push({
                        ruta: nuevaRuta,
                        descripcion: "Clave solo en primer objeto"
                    });
                } else {
                    this.compararRecursivo(obj1[key], obj2[key], nuevaRuta, diferencias);
                }
            });
        } else if (obj1 !== obj2) {
            diferencias.push({
                ruta: ruta || "raíz",
                descripcion: `${obj1} vs ${obj2}`
            });
        }
    }
}

📝 Resumen Final

📄 JSON es fundamental en el desarrollo web moderno. Dominar su uso, validación y manejo de errores es esencial para crear aplicaciones robustas que intercambien datos de manera eficiente y segura.

🎯 Puntos clave a recordar:

  • Sintaxis estricta: Comillas dobles, sin comentarios, sin funciones
  • Manejo de errores: Siempre usar try/catch con JSON.parse()
  • Transformaciones: Usar replacer y reviver para filtrar y transformar datos
  • Validación: Verificar estructura y tipos antes de procesar
  • Seguridad: Filtrar datos sensibles antes de serializar
