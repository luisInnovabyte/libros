Promises en JavaScript

Las Promises son una característica fundamental de JavaScript moderno para manejar operaciones asíncronas de manera más elegante y legible que los callbacks tradicionales.

💡 Concepto clave: Una Promise representa un valor que puede estar disponible ahora, en el futuro, o nunca. Es como un "contrato" que garantiza que se ejecutará alguna acción cuando se complete una tarea asíncrona.

¿Por qué necesitamos Promises?

  • 📚 Legibilidad: Código más limpio y fácil de entender
  • 🔧 Mantenibilidad: Más fácil de modificar y debuggear
  • 🎯 Control de errores: Manejo centralizado de errores con .catch()
  • 🔄 Composición: Fácil combinación de múltiples operaciones asíncronas
  • ⚡ Rendimiento: Mejor gestión de operaciones paralelas

⚖️ Operaciones Síncronas vs Asíncronas

Comprende la diferencia fundamental entre código que bloquea y código que no bloquea.

🔒 Operaciones Síncronas

// Código síncrono - bloquea la ejecución
console.log("1. Inicio");

function operacionSincrona() {
    console.log("2. Procesando...");
    
    // Simulación de trabajo pesado (bloquea el hilo)
    for (let i = 0; i < 1000000000; i++) {
        // Operación que consume tiempo
    }
    
    console.log("3. Procesamiento completado");
    return "Resultado síncrono";
}

const resultado = operacionSincrona(); // Espera aquí
console.log("4. Resultado:", resultado);
console.log("5. Fin");

// Salida:
// 1. Inicio
// 2. Procesando...
// 3. Procesamiento completado  
// 4. Resultado: Resultado síncrono
// 5. Fin
  • ✅ Predecibles: orden exacto de ejecución
  • ✅ Simples de entender: flujo lineal
  • ❌ Bloquean el hilo principal
  • ❌ Ineficientes para operaciones lentas

🔓 Operaciones Asíncronas

// Código asíncrono - no bloquea
console.log("1. Inicio");

function operacionAsincrona() {
    console.log("2. Iniciando operación asíncrona...");
    
    // setTimeout es asíncrono - no bloquea
    setTimeout(() => {
        console.log("4. Operación asíncrona completada");
    }, 2000);
    
    console.log("3. Operación iniciada, continuando...");
}

operacionAsincrona(); // No bloquea aquí
console.log("5. Fin del programa principal");

// Salida:
// 1. Inicio
// 2. Iniciando operación asíncrona...
// 3. Operación iniciada, continuando...
// 5. Fin del programa principal
// 4. Operación asíncrona completada (después de 2s)
  • ✅ No bloquean: el programa continúa
  • ✅ Eficientes: ideales para operaciones lentas
  • ✅ Mejor UX: la interfaz no se congela
  • ❌ Más complejas: requieren manejo especial
⚠️ El problema del "Callback Hell": Antes de las Promises, las operaciones asíncronas se manejaban con callbacks anidados, creando código difícil de leer y mantener.

🤝 ¿Qué son las Promises?

Una Promise es un objeto que representa la eventual finalización (o falla) de una operación asíncrona y su valor resultante.

📊 Estados de una Promise

⏳ Pending (Pendiente)

Estado inicial, ni cumplida ni rechazada

✅ Fulfilled (Cumplida)

La operación se completó exitosamente

❌ Rejected (Rechazada)

La operación falló

// Ejemplo visual de los estados
const promiseEjemplo = new Promise((resolve, reject) => {
    console.log("Estado: Pending"); // Se ejecuta inmediatamente
    
    setTimeout(() => {
        const exito = Math.random() > 0.5;
        
        if (exito) {
            resolve("¡Operación exitosa!");     // Estado: Fulfilled
        } else {
            reject("Error en la operación");    // Estado: Rejected
        }
    }, 2000);
});

promiseEjemplo
    .then(resultado => console.log("Estado: Fulfilled -", resultado))
    .catch(error => console.log("Estado: Rejected -", error));

Aprende a crear y consumir Promises con la sintaxis correcta.

🔧 Crear una Promise

// Sintaxis básica
const miPromise = new Promise((resolve, reject) => {
    // Lógica asíncrona aquí
    
    // Si todo sale bien:
    // resolve(valor);
    
    // Si algo sale mal:
    // reject(error);
});

// Ejemplo práctico: simular una API
function obtenerDatosUsuario(id) {
    return new Promise((resolve, reject) => {
        // Simular tiempo de respuesta de API
        setTimeout(() => {
            if (id > 0) {
                const usuario = {
                    id: id,
                    nombre: `Usuario ${id}`,
                    email: `usuario${id}@email.com`,
                    activo: true
                };
                resolve(usuario); // Operación exitosa
            } else {
                reject(new Error("ID de usuario inválido"));
            }
        }, 1500);
    });
}

📞 Consumir Promises

// Usar la promise con then/catch
obtenerDatosUsuario(123)
    .then(usuario => {
        console.log("Usuario obtenido:", usuario);
        // { id: 123, nombre: "Usuario 123", ... }
    })
    .catch(error => {
        console.error("Error:", error.message);
    })
    .finally(() => {
        console.log("Operación completada");
    });

// Caso con error
obtenerDatosUsuario(-1)
    .then(usuario => {
        console.log("No se ejecutará");
    })
    .catch(error => {
        console.error("Error capturado:", error.message);
        // "ID de usuario inválido"
    });

🔧 Métodos de Promise

Métodos estáticos útiles para trabajar con Promises.

⚡ Promise.resolve() y Promise.reject()

// Promise.resolve() - crear promise ya resuelta
const promiseResuelta = Promise.resolve("Valor inmediato");
promiseResuelta.then(valor => console.log(valor));

// Promise.reject() - crear promise ya rechazada
const promiseRechazada = Promise.reject("Error inmediato");
promiseRechazada.catch(error => console.log(error));

// Ejemplo práctico: validación
function validarEmail(email) {
    const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    
    if (regex.test(email)) {
        return Promise.resolve({
            email: email,
            valido: true,
            mensaje: "Email válido"
        });
    } else {
        return Promise.reject({
            email: email,
            valido: false,
            mensaje: "Email inválido"
        });
    }
}

📋 Promise.allSettled()

// Promise.allSettled() espera a que todas terminen
function procesarPedido(id, fallaEn = null) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            if (id === fallaEn) {
                reject(`Pedido ${id} falló`);
            } else {
                resolve(`Pedido ${id} procesado correctamente`);
            }
        }, Math.random() * 2000 + 500);
    });
}

const pedidos = [
    procesarPedido(1001),
    procesarPedido(1002, 1002), // Este fallará
    procesarPedido(1003),
    procesarPedido(1004, 1004), // Este también fallará
    procesarPedido(1005)
];

Promise.allSettled(pedidos)
    .then(resultados => {
        resultados.forEach((resultado, index) => {
            const pedidoId = 1001 + index;
            if (resultado.status === 'fulfilled') {
                console.log(`✅ Pedido ${pedidoId}: ${resultado.value}`);
            } else {
                console.log(`❌ Pedido ${pedidoId}: ${resultado.reason}`);
            }
        });
    });

🚀 Promise.all() - Operaciones en Paralelo

Ejecuta múltiples Promises simultáneamente y espera a que todas se completen.

💡 Importante: Promise.all() falla completamente si cualquier Promise es rechazada. Todas las operaciones se ejecutan en paralelo, no secuencialmente. 
// Ejemplo: descargar múltiples archivos en paralelo
function descargarArchivo(nombre, tamaño) {
    return new Promise((resolve) => {
        const tiempoDescarga = tamaño * 100; // Simular tiempo basado en tamaño
        
        setTimeout(() => {
            resolve({
                archivo: nombre,
                tamaño: tamaño,
                descargado: true,
                tiempo: tiempoDescarga
            });
        }, tiempoDescarga);
    });
}

// Descargar múltiples archivos en paralelo
const descargas = [
    descargarArchivo("documento.pdf", 5),
    descargarArchivo("imagen.jpg", 3),
    descargarArchivo("video.mp4", 15)
];

console.log("🚀 Iniciando descargas en paralelo...");
const inicioDescargas = Date.now();

Promise.all(descargas)
    .then(resultados => {
        const tiempoTotal = Date.now() - inicioDescargas;
        console.log("✅ Todas las descargas completadas");
        
        resultados.forEach(resultado => {
            console.log(`📁 ${resultado.archivo} (${resultado.tamaño}MB) - ${resultado.tiempo}ms`);
        });
        
        console.log(`⏱️ Tiempo total: ${tiempoTotal}ms`);
    })
    .catch(error => {
        console.error("❌ Error en alguna descarga:", error);
    });

🏁 Promise.race() - La Primera que Termine

Se resuelve con la primera Promise que se complete, ya sea exitosa o con error.

// Ejemplo: competencia entre servidores
function servidor(nombre, latencia) {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve({
                servidor: nombre,
                datos: `Datos desde ${nombre}`,
                latencia: latencia
            });
        }, latencia);
    });
}

// Competencia entre servidores
const servidores = [
    servidor("Servidor-US", 800),
    servidor("Servidor-EU", 600),
    servidor("Servidor-ASIA", 1200)
];

console.log("🏁 Iniciando competencia de servidores...");

Promise.race(servidores)
    .then(ganador => {
        console.log("🏆 Servidor más rápido:", ganador.servidor);
        console.log("📡 Datos:", ganador.datos);
        console.log("⚡ Latencia:", ganador.latencia + "ms");
    });

// Timeout con Promise.race()
function operacionLenta() {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("Operación completada después de 5 segundos");
        }, 5000);
    });
}

function timeout(ms) {
    return new Promise((_, reject) => {
        setTimeout(() => {
            reject(new Error(`Timeout después de ${ms}ms`));
        }, ms);
    });
}

// Limitar tiempo máximo de espera
Promise.race([
    operacionLenta(),
    timeout(3000) // Timeout de 3 segundos
])
.then(resultado => {
    console.log("✅ Completado:", resultado);
})
.catch(error => {
    console.log("⏰ Timeout:", error.message);
});

🔗 Cadenas de Promises (Promise Chaining)

Encadena operaciones asíncronas de manera secuencial y elegante.

// Ejemplo: proceso de registro de usuario
function validarDatos(datos) {
    return new Promise((resolve, reject) => {
        console.log("1️⃣ Validando datos...");
        
        setTimeout(() => {
            if (datos.email && datos.password && datos.password.length >= 6) {
                resolve({ ...datos, validado: true });
            } else {
                reject(new Error("Datos inválidos"));
            }
        }, 500);
    });
}

function verificarEmailUnico(userData) {
    return new Promise((resolve, reject) => {
        console.log("2️⃣ Verificando email único...");
        
        setTimeout(() => {
            const emailExiste = userData.email === "admin@test.com";
            
            if (!emailExiste) {
                resolve({ ...userData, emailUnico: true });
            } else {
                reject(new Error("Email ya registrado"));
            }
        }, 800);
    });
}

function crearUsuario(userData) {
    return new Promise((resolve) => {
        console.log("3️⃣ Creando usuario en base de datos...");
        
        setTimeout(() => {
            const nuevoUsuario = {
                ...userData,
                id: Math.random().toString(36).substr(2, 9),
                fechaRegistro: new Date(),
                activo: true
            };
            resolve(nuevoUsuario);
        }, 1000);
    });
}

// Cadena de promises para registro completo
const datosRegistro = {
    nombre: "Juan Pérez",
    email: "juan@example.com",
    password: "123456789"
};

validarDatos(datosRegistro)
    .then(datosValidados => {
        console.log("✅ Datos válidos");
        return verificarEmailUnico(datosValidados);
    })
    .then(datosVerificados => {
        console.log("✅ Email único");
        return crearUsuario(datosVerificados);
    })
    .then(usuarioCreado => {
        console.log("✅ Usuario creado con ID:", usuarioCreado.id);
        return usuarioCreado;
    })
    .catch(error => {
        console.error("❌ Error en el registro:", error.message);
    })
    .finally(() => {
        console.log("🔄 Proceso de registro finalizado");
    });

❌ Manejo de Errores con Promises

Estrategias para manejar errores en cadenas de Promises y operaciones asíncronas.

🛑 Error Detiene la Cadena

function paso1() {
    return Promise.resolve("Paso 1 completado");
}

function paso2() {
    return Promise.reject(new Error("Error en paso 2"));
}

function paso3() {
    return Promise.resolve("Paso 3 completado");
}

// El error en paso2 detiene toda la cadena
paso1()
    .then(resultado1 => {
        console.log("✅", resultado1);
        return paso2(); // Este fallará
    })
    .then(resultado2 => {
        console.log("✅", resultado2); // No se ejecutará
        return paso3();
    })
    .then(resultado3 => {
        console.log("✅", resultado3); // No se ejecutará
    })
    .catch(error => {
        console.error("❌ Error capturado:", error.message);
    });

🔧 Recuperación de Errores

// Manejar errores específicos y continuar
function procesarConRecuperacion() {
    return paso1()
        .then(resultado1 => {
            console.log("✅", resultado1);
            return paso2().catch(error => {
                console.log("⚠️ Error en paso 2, usando valor por defecto");
                return "Valor por defecto para paso 2";
            });
        })
        .then(resultado2 => {
            console.log("✅", resultado2); // Se ejecutará
            return paso3();
        })
        .then(resultado3 => {
            console.log("✅", resultado3); // Se ejecutará
        })
        .catch(error => {
            console.error("❌ Error final:", error.message);
        });
}

procesarConRecuperacion();
💡 Tip: Usa .catch() en puntos específicos de la cadena para manejar errores localmente, o al final para manejo global.

Implementaciones reales y útiles usando Promises.

💾 Sistema de Cache con Promises

class CacheConPromises {
    constructor(ttl = 60000) { // TTL: 1 minuto
        this.cache = new Map();
        this.ttl = ttl;
    }
    
    obtener(clave, funcionCarga) {
        const item = this.cache.get(clave);
        
        // Verificar si existe y no ha expirado
        if (item && Date.now() - item.timestamp < this.ttl) {
            console.log(`📋 Cache HIT para ${clave}`);
            return Promise.resolve(item.valor);
        }
        
        console.log(`🔍 Cache MISS para ${clave}, cargando...`);
        
        // Cargar datos y cachear
        return funcionCarga()
            .then(valor => {
                this.cache.set(clave, {
                    valor: valor,
                    timestamp: Date.now()
                });
                console.log(`💾 Datos cacheados para ${clave}`);
                return valor;
            });
    }
}

// Uso del cache
const cache = new CacheConPromises(5000);

function consultarAPI(endpoint) {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve({
                endpoint: endpoint,
                datos: `Datos de ${endpoint}`,
                timestamp: new Date().toISOString()
            });
        }, 2000);
    });
}

🔄 Retry Automático

function retry(funcion, maxIntentos = 3, delay = 1000) {
    return new Promise((resolve, reject) => {
        let intentos = 0;
        
        function ejecutarIntento() {
            intentos++;
            console.log(`🔄 Intento ${intentos}/${maxIntentos}`);
            
            funcion()
                .then(resolve) // Si tiene éxito, resolvemos
                .catch(error => {
                    console.log(`❌ Intento ${intentos} falló:`, error.message);
                    
                    if (intentos >= maxIntentos) {
                        reject(new Error(`Falló después de ${maxIntentos} intentos`));
                    } else {
                        console.log(`⏰ Reintentando en ${delay}ms...`);
                        setTimeout(ejecutarIntento, delay);
                    }
                });
        }
        
        ejecutarIntento();
    });
}

// Función que falla ocasionalmente
function operacionInestable() {
    return new Promise((resolve, reject) => {
        const exito = Math.random() > 0.7; // 30% probabilidad
        
        if (exito) {
            resolve("¡Operación exitosa!");
        } else {
            reject(new Error("Falla temporal del servicio"));
        }
    });
}

// Uso del retry
retry(operacionInestable, 5, 1500)
    .then(resultado => console.log("✅ Éxito final:", resultado))
    .catch(error => console.log("🔥 Fallo definitivo:", error.message));

✨ Mejores Prácticas

Consejos para escribir código con Promises limpio y eficiente.

✅ Buenas Prácticas

// 1. Siempre manejar errores
function buenaPractica1() {
    return fetch('/api/datos')
        .then(response => response.json())
        .then(datos => procesarDatos(datos))
        .catch(error => {
            console.error('Error:', error);
            throw error; // Re-lanzar si es necesario
        });
}

// 2. Usar Promise.all() para operaciones paralelas
function buenaPractica2() {
    const promises = [
        obtenerDatosUsuario(),
        obtenerConfiguracion(),
        obtenerPermisos()
    ];
    
    return Promise.all(promises)
        .then(([usuario, config, permisos]) => {
            return { usuario, config, permisos };
        });
}

// 3. Crear promises reutilizables
function crearTimeout(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

function conTimeout(promise, ms) {
    return Promise.race([
        promise,
        crearTimeout(ms).then(() => 
            Promise.reject(new Error('Timeout'))
        )
    ]);
}

❌ Prácticas a Evitar

// 1. NO crear promises innecesarias
// ❌ Malo
function malasPracticas1() {
    return new Promise((resolve) => {
        resolve(valor); // Innecesario
    });
}
// ✅ Mejor
function buena1() {
    return Promise.resolve(valor);
}

// 2. NO anidar promises (Promise hell)
// ❌ Malo
function malasPracticas2() {
    return obtenerDatos()
        .then(datos => {
            return procesarDatos(datos)
                .then(resultado => {
                    return guardarResultado(resultado);
                });
        });
}
// ✅ Mejor
function buena2() {
    return obtenerDatos()
        .then(datos => procesarDatos(datos))
        .then(resultado => guardarResultado(resultado));
}

// 3. NO olvidar return en las cadenas
// ❌ Malo
function malasPracticas3() {
    return obtenerDatos()
        .then(datos => {
            procesarDatos(datos); // Falta return!
        });
}
// ✅ Mejor
function buena3() {
    return obtenerDatos()
        .then(datos => {
            return procesarDatos(datos);
        });
}
🚀 Siguiente paso: Una vez que domines las Promises, puedes avanzar a async/await que proporciona una sintaxis aún más limpia para trabajar con código asíncrono.

📚 Resumen

Las Promises son fundamentales en JavaScript moderno para:

  • Manejar operaciones asíncronas: Sin bloquear el hilo principal
  • Evitar callback hell: Código más limpio y legible
  • Composición de operaciones: Cadenas y operaciones paralelas
  • Manejo robusto de errores: Control centralizado
  • Base para async/await: Sintaxis moderna
